# Site Scanner

# The application. Everything below /app is client-side state — a drawn area,
# a chosen scanner, an open investigation — held in a URL fragment that never
# reaches the server. There is nothing here for a crawler to index, and an
# assessment URL in search results would be someone else's site.
User-agent: *
Disallow: /app

# The thank-you page is reachable only after submitting the form. In search
# results it is a page people arrive at without having done the thing it
# thanks them for.
Disallow: /thank-you

# Everything else is public and should be found.
Allow: /

# production domain exists.
